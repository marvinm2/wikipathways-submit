{% block script %}
  async function setItem(pr, key, state) {
    const fd = new FormData(); fd.append('key', key); fd.append('state', state);
    const r = await fetch('/api/reviews/' + pr + '/checklist', {method:'POST', body:fd});
    if (r.ok) location.reload();
    else showMsg(pr, 'Error ' + r.status);
  }
  async function approve(pr) {
    const r = await fetch('/api/reviews/' + pr + '/approve', {method:'POST'});
    const j = await r.json();
    if (r.ok) location.reload();
    else showMsg(pr, 'Error ' + r.status + ': ' + JSON.stringify(j.detail));
  }
  function showMsg(pr, text) {
    const el = document.querySelector('[data-pr="' + pr + '"] .msg');
    el.hidden = false; el.textContent = text;
  }
